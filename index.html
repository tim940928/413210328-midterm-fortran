! 程式名稱：計算 1 到 N 的總和
! 作者：許耕綸
! 功能：讀取使用者輸入的正整數 N，計算並輸出 1+2+...+N 的結果

program sum_1_to_n
    implicit none          ! 強制所有變數必須明確宣告型別
    integer :: n           ! 使用者輸入的正整數 N
    integer :: i           ! 迴圈計數變數
    integer :: total       ! 用來累加總和的變數

    ! 提示使用者輸入
    print *, "請輸入正整數 N："
    
    ! 讀取使用者輸入的 N
    read *, n

    ! 初始化總和為 0
    total = 0
    
    ! 使用 do 迴圈從 1 累加到 N
    do i = 1, n
        total = total + i  ! 將目前的 i 加到總和中
    end do

    ! 輸出最終結果
    print *, "1 到", n, "的總和為：", total
end program sum_1_to_n
